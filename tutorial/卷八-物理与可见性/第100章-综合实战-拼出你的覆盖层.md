## 主循环的最终形态

```cpp
int main(int argc, char *argv[]) {
    // ===== 初始化 =====
    ConfigManager::LoadConfig();                    // 卷三：读配置

    graphics = GraphicsManager::getGraphicsInterface();   // 卷五：Vulkan 后端

    screen_config();                                     // 卷五：屏幕信息
    native_window_screen_x = displayInfo.width;
    native_window_screen_y = displayInfo.height;
    abs_ScreenX = displayInfo.width;
    abs_ScreenY = displayInfo.height;

    window = ANativeWindowCreator::Create(                // 卷五：建图层
        "overlay", native_window_screen_x, native_window_screen_y);
    graphics->Init_Render(window, native_window_screen_x, native_window_screen_y);

    Touch::Init({(float)abs_ScreenX, (float)abs_ScreenY}, true);   // 卷五
    Touch::setOrientation(displayInfo.orientation);

    init_My_drawdata();                                  // 卷五：字体与样式

    // ===== 主循环 =====
    bool running = true;
    while (running) {
        drawBegin();                                     // 卷五：帧率+尺寸
        Touch::UpdateImGuiInput();                       // 卷五：触摸
        graphics->NewFrame(true);

        Layout_tick_UI(&running);                        // 卷五：菜单
        UpdateGameData();                                // 卷七：读数据
        UpdateCamera();                                  // 卷七：相机
        BuildProjectionMatrix();                         // 卷四：投影

        auto *draw = ImGui::GetForegroundDrawList();
        DrawPlayers(draw);                               // 卷四+卷七：ESP
        DrawPhysicsWireframe(draw);                      // 卷八：可见几何

        graphics->EndFrame();                            // 卷五：提交
    }

    // ===== 清理 =====
    Touch::Close();
    graphics->Shutdown();
    ANativeWindowCreator::Destroy(window);
    ConfigManager::SaveConfig();
    return 0;
}
```

## 核心绘制流程（串起所有卷）

```cpp
void DrawPlayers(ImDrawList *draw) {
    // ① 一次性读完整个 Actor 数组          卷六（批量读）
    std::vector<uint64_t> actors(count);
    dr->Read(arrayAddr, actors.data(), count * 8);

    // ② 批量读每个对象的字段              卷六
    ReadBatch(dr, requests);

    for (auto &snapshot : snapshots) {
        // ③ 合法性过滤                    卷七（第89章）
        if (!IsValidObject(snapshot.address)) continue;

        // ④ 名字解析 + 类型过滤            卷七（第85章）
        std::string name = GetNameById(snapshot.nameId);
        if (name.find("BP_Character") == npos) continue;

        // ⑤ 世界坐标 → 屏幕坐标            卷四（第51章）
        Vec2 screen;
        if (!WorldToScreen(snapshot.position, screen)) continue;

        // ⑥ 遮挡判定                      卷八（第95章）
        bool occluded = physicsReady && IsOccluded(camPos, aimPoint);

        // ⑦ 画方框                        卷五（第61章）
        draw->AddRect(min, max, occluded ? GREEN : RED);

        // ⑧ 画骨骼                        卷七（第86章）
        if (showBones) {
            for (int seg = 0; seg < 14; seg++)
                draw->AddLine(bones[segA[seg]], bones[segB[seg]], color);
        }
    }
}
```
