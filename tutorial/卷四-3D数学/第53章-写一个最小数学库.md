# 第 53 章 · 写一个最小数学库

> [!abstract] 本章目标
> 完成总纲承诺的第 3 个成果：**`mathlib`**——
> 一个自包含的 3D 数学库，后面所有章节都基于它。

> [!note] 承上
> 前 8 章的数学都是散在各章里的内联代码。
> 本章把它们**沉淀成一个库** `mathlib`（总纲承诺的第 3 个成果），后面所有章节都基于它。

## 先看它要提供什么

| 类型 | 运算 |
|---|---|
| `Vec2` | 加减、数乘、点积、长度 |
| `Vec3` | 加减、数乘、点积、叉积、长度、归一化、距离 |
| `Rotator` | 度/弧度转换、角度归一化、角度差 |
| `Quat` | 与轴角互转、转矩阵 |
| `Mat4` | 单位、平移、缩放、旋转、乘法、转置、取位置 |
| 相机 | `MakeViewProj`、`WorldToScreen` |

## 设计原则

1. **头文件实现**（`inline`），方便引入（第 09 章）
2. **全部 `constexpr`/`inline`**，零运行时开销
3. **防除零、防 NaN**（第 26 章）
4. **命名清晰**：`HalfWidth` 不叫 `px`

## 完整的 mathlib.h

```cpp
// mathlib.h —— 最小 3D 数学库
#pragma once
#include <cmath>
#include <algorithm>
#include <limits>

namespace ml {

constexpr float PI      = 3.14159265358979323846f;
constexpr float DEG2RAD = PI / 180.0f;
constexpr float RAD2DEG = 180.0f / PI;
constexpr float EPS     = 1e-6f;

inline bool IsFinite(float v) { return std::isfinite(v); }

// ---------- Vec2 ----------
struct Vec2 {
    float x = 0, y = 0;
    constexpr Vec2() = default;
    constexpr Vec2(float x_, float y_) : x(x_), y(y_) {}

    constexpr Vec2 operator+(const Vec2& v) const { return {x+v.x, y+v.y}; }
    constexpr Vec2 operator-(const Vec2& v) const { return {x-v.x, y-v.y}; }
    constexpr Vec2 operator*(float s)       const { return {x*s,   y*s  }; }
    constexpr Vec2 operator/(float s)       const { return {x/s,   y/s  }; }

    constexpr float Dot(const Vec2& v)  const { return x*v.x + y*v.y; }
    constexpr float LengthSq()          const { return x*x + y*y; }
    float Length()                      const { return std::sqrt(LengthSq()); }
    Vec2  Normalized()                  const {
        float l = Length();
        return l < EPS ? Vec2{} : (*this) / l;
    }
    bool  IsValid() const { return IsFinite(x) && IsFinite(y); }
};

// ---------- Vec3 ----------
struct Vec3 {
    float x = 0, y = 0, z = 0;
    constexpr Vec3() = default;
    constexpr Vec3(float x_, float y_, float z_) : x(x_), y(y_), z(z_) {}

    constexpr Vec3 operator+(const Vec3& v) const { return {x+v.x, y+v.y, z+v.z}; }
    constexpr Vec3 operator-(const Vec3& v) const { return {x-v.x, y-v.y, z-v.z}; }
    constexpr Vec3 operator*(float s)       const { return {x*s,   y*s,   z*s  }; }
    constexpr Vec3 operator/(float s)       const { return {x/s,   y/s,   z/s  }; }

    constexpr float Dot(const Vec3& v) const { return x*v.x + y*v.y + z*v.z; }
    constexpr Vec3  Cross(const Vec3& v) const {
        return {y*v.z - z*v.y, z*v.x - x*v.z, x*v.y - y*v.x};
    }
    constexpr float LengthSq() const { return x*x + y*y + z*z; }
    float Length()             const { return std::sqrt(LengthSq()); }

    Vec3 Normalized() const {
        float l = Length();
        return l < EPS ? Vec3{} : (*this) / l;
    }

    static float Distance(const Vec3& a, const Vec3& b) {
        return (b - a).Length();
    }
    static float DistanceSq(const Vec3& a, const Vec3& b) {
        return (b - a).LengthSq();
    }
    bool IsValid() const { return IsFinite(x) && IsFinite(y) && IsFinite(z); }
};

// ---------- Rotator (度) ----------
struct Rotator {
    float Pitch = 0, Yaw = 0, Roll = 0;   // 度
};

// 角度归一化到 (-180, 180]
inline float NormalizeAngle(float deg) {
    while (deg >  180.0f) deg -= 360.0f;
    while (deg <= -180.0f) deg += 360.0f;
    return deg;
}

// 角度差（走最短路径）
inline float AngleDelta(float from, float to) {
    return NormalizeAngle(to - from);
}

// 方向 → 欧拉角（UE4 约定：Z 向上）
inline Rotator DirectionToRotator(const Vec3& dir) {
    const float horiz = std::sqrt(dir.x*dir.x + dir.y*dir.y);
    Rotator r;
    r.Pitch = std::atan2(dir.z, horiz) * RAD2DEG;
    r.Yaw   = std::atan2(dir.y, dir.x) * RAD2DEG;
    r.Roll  = 0.0f;
    return r;
}

// 欧拉角 → 前方向（UE4 约定）
inline Vec3 RotatorToForward(const Rotator& r) {
    const float sp = std::sin(r.Pitch * DEG2RAD), cp = std::cos(r.Pitch * DEG2RAD);
    const float sy = std::sin(r.Yaw   * DEG2RAD), cy = std::cos(r.Yaw   * DEG2RAD);
    return {cp*cy, cp*sy, sp};
}

// ---------- Quat ----------
struct Quat {
    float x = 0, y = 0, z = 0, w = 1;
};

// 轴 + 角度（度）→ 四元数
inline Quat QuatFromAxisAngle(const Vec3& axis, float degrees) {
    const float half = degrees * DEG2RAD * 0.5f;
    const float s = std::sin(half);
    Vec3 n = axis.Normalized();
    return {n.x*s, n.y*s, n.z*s, std::cos(half)};
}

// ---------- Mat4（行主序，M[row][col]）----------
// 注意：本库有两种矩阵表示，别混——
//   · Mat4 结构体：行主序、二维 M[row][col]，用于自己写变换、骨骼、测试（直观）
//   · Camera 里的 M[16] 一维数组：列主序存储 M[row + col*4]，用于对齐项目源码 camMakeMatrix
// 这是故意的：前者是"干净的数学"，后者是"照着真实项目抄"。第 46/49 章讲过两者区别。
struct Mat4 {
    float M[4][4] = {};

    static Mat4 Identity() {
        Mat4 m;
        for (int i = 0; i < 4; i++) m.M[i][i] = 1.0f;
        return m;
    }
    float* operator[](int i) { return M[i]; }
    const float* operator[](int i) const { return M[i]; }
};

inline Mat4 Mat4Multiply(const Mat4& a, const Mat4& b) {
    Mat4 r;
    for (int i = 0; i < 4; i++)
        for (int j = 0; j < 4; j++)
            for (int k = 0; k < 4; k++)
                r.M[i][j] += a.M[i][k] * b.M[k][j];
    return r;
}

inline Mat4 Mat4Translation(const Vec3& t) {
    Mat4 m = Mat4::Identity();
    m.M[3][0] = t.x; m.M[3][1] = t.y; m.M[3][2] = t.z;
    return m;
}

inline Mat4 Mat4Scale(const Vec3& s) {
    Mat4 m = Mat4::Identity();
    m.M[0][0] = s.x; m.M[1][1] = s.y; m.M[2][2] = s.z;
    return m;
}

inline Mat4 Mat4Transpose(const Mat4& m) {
    Mat4 r;
    for (int i = 0; i < 4; i++)
        for (int j = 0; j < 4; j++)
            r.M[i][j] = m.M[j][i];
    return r;
}

inline Vec3 Mat4GetPosition(const Mat4& m) {
    return {m.M[3][0], m.M[3][1], m.M[3][2]};
}

// 四元数 + 位置 + 缩放 → 矩阵
inline Mat4 QuatToMatrix(const Quat& q, const Vec3& pos, const Vec3& scale) {
    Mat4 m;
    const float x2 = q.x + q.x, y2 = q.y + q.y, z2 = q.z + q.z;
    const float xx2 = q.x*x2, yy2 = q.y*y2, zz2 = q.z*z2;
    const float yz2 = q.y*z2, wx2 = q.w*x2;
    const float xy2 = q.x*y2, wz2 = q.w*z2;
    const float xz2 = q.x*z2, wy2 = q.w*y2;

    m.M[0][0] = (1 - (yy2 + zz2)) * scale.x;
    m.M[1][1] = (1 - (xx2 + zz2)) * scale.y;
    m.M[2][2] = (1 - (xx2 + yy2)) * scale.z;

    m.M[2][1] = (yz2 - wx2) * scale.z;
    m.M[1][2] = (yz2 + wx2) * scale.y;

    m.M[1][0] = (xy2 - wz2) * scale.y;
    m.M[0][1] = (xy2 + wz2) * scale.x;

    m.M[2][0] = (xz2 + wy2) * scale.z;
    m.M[0][2] = (xz2 - wy2) * scale.x;

    m.M[3][0] = pos.x; m.M[3][1] = pos.y; m.M[3][2] = pos.z;
    m.M[3][3] = 1.0f;
    return m;
}

// ---------- 相机 ----------
struct Camera {
    Vec3    Position;
    Rotator Rotation;
    float   FOV = 90.0f;

    float HalfWidth  = 0;    // 屏幕半宽（像素）
    float HalfHeight = 0;    // 屏幕半高
    float Kx = 1.0f, Ky = 1.0f;   // 焦距微调

    float M[16] = {};        // 列主序存储：M[row + col*4]

    bool BuildMatrix() {
        if (FOV < 1.0f || FOV > 179.0f) return false;
        if (!Position.IsValid()) return false;
        if (Position.LengthSq() < EPS) return false;   // 位置为 0，数据无效

        const float pitch = Rotation.Pitch * DEG2RAD;
        const float yaw   = Rotation.Yaw   * DEG2RAD;
        const float sp = std::sin(pitch), cp = std::cos(pitch);
        const float sy = std::sin(yaw),   cy = std::cos(yaw);
        float t = std::tan(FOV * 0.5f * DEG2RAD);
        if (std::fabs(t) < EPS) t = (t >= 0 ? EPS : -EPS);

        // 深度行
        M[3]  = cy*cp;  M[7]  = sy*cp;  M[11] = sp;
        M[15] = -(Position.x*M[3] + Position.y*M[7] + Position.z*M[11]);

        // right 行
        M[0]  = -sy/t;  M[4]  = cy/t;   M[8]  = 0.0f;
        M[12] = -(Position.x*(-sy) + Position.y*cy) / t;

        // up 行
        M[1]  = (-sp*cy)/t;  M[5] = (-sp*sy)/t;  M[9] = cp/t;
        M[13] = -(Position.x*(-sp*cy) + Position.y*(-sp*sy) + Position.z*cp) / t;

        M[2] = M[6] = M[10] = M[14] = 0.0f;
        return true;
    }

    // 世界坐标 → 屏幕坐标；失败返回 false
    bool WorldToScreen(const Vec3& world, Vec2& out) const {
        const float w = M[3]*world.x + M[7]*world.y + M[11]*world.z + M[15];
        if (w < 0.001f) return false;

        const float cx = M[0]*world.x + M[4]*world.y + M[8] *world.z + M[12];
        const float cy = M[1]*world.x + M[5]*world.y + M[9] *world.z + M[13];

        out.x = (cx / w * Kx + 1.0f) * HalfWidth;
        out.y = (1.0f - cy / w * Ky) * HalfHeight;
        return IsFinite(out.x) && IsFinite(out.y);
    }

    // 是否在屏幕范围内（可选留白）
    static bool OnScreen(const Vec2& p, float w, float h, float margin = 0) {
        return p.x >= -margin && p.x <= w + margin &&
               p.y >= -margin && p.y <= h + margin;
    }
};

} // namespace ml
```

## 用法示例

> [!important] 先保存 mathlib.h 再试下面的例子
> 把上面"完整的 mathlib.h"**整段复制，存成 `mathlib.h`**（放在你练习目录里），
> 再新建下面的 `main.cpp` 和它放**同一目录**。否则会报 `fatal error: mathlib.h: No such file or directory`。
> 编译：`g++ -std=c++17 main.cpp -o main`。

```cpp
#include "mathlib.h"
#include <cstdio>

int main(void) {
    using namespace ml;

    Camera cam;
    cam.Position  = {0, 0, 50};
    cam.Rotation  = {0, 0, 0};
    cam.FOV       = 90.0f;
    cam.HalfWidth = 1080.0f;
    cam.HalfHeight= 1200.0f;

    if (!cam.BuildMatrix()) { printf("相机构建失败\n"); return 1; }

    Vec2 s;
    Vec3 targets[] = {
        {100, 0, 50},      // 正前方
        {100, 50, 50},     // 右前
        {-100, 0, 50},     // 后方
    };
    for (const auto& t : targets) {
        if (cam.WorldToScreen(t, s))
            printf("(%.0f,%.0f,%.0f) → (%.1f, %.1f)\n", t.x,t.y,t.z, s.x, s.y);
        else
            printf("(%.0f,%.0f,%.0f) → 不可见\n", t.x,t.y,t.z);
    }
    return 0;
}
```

## 动手：扩展 mathlib

给它加三个常用函数：

```cpp
// 1. 线性插值
inline float Lerp(float a, float b, float t) { return a + (b - a) * t; }
inline Vec3  Lerp(const Vec3& a, const Vec3& b, float t) {
    return {Lerp(a.x,b.x,t), Lerp(a.y,b.y,t), Lerp(a.z,b.z,t)};
}

// 2. 角度插值（走最短路径）—— 自瞄平滑会用到
inline float LerpAngle(float a, float b, float t) {
    return NormalizeAngle(a + AngleDelta(a, b) * t);
}

// 3. 限幅
inline float Clamp(float v, float lo, float hi) {
    return std::clamp(v, lo, hi);
}
```

然后写个小程序验证：从 350° 插值到 10°，中间值应该是 350→355→0→5→10，
**而不是** 350→300→...→10（绕远路）。

## 常见坑排查表

| 症状 | 最可能的原因 | 解法 |
|---|---|---|
| 编译报 `mathlib.h not found` | 没保存 `mathlib.h` 或不在同目录 | 把完整代码存成 `mathlib.h`，和 `main.cpp` 同目录 |
| 归一化产生 **NaN** | 零向量归一化没防除零 | `if (l < EPS) return {};` |
| 两种矩阵表示**混用出错** | 没分清 `Matrix`（行主序）和 `matrix[16]`（列主序）| 本项目两种并存，看上下文 |
| 角度插值**绕远路** | 没做角度归一化 | 用 `AngleDelta` 归一到 ±180° |
